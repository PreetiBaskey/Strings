//approach - 1
//time complexity - O(n)
//space complexity - O(1)
class Solution {
public:
    bool isOnly_1(string s) {
        int n = s.length();
        for(int i = 0; i < n; i++) {
            if(s[i] != '1') {
                return false;
            }
        }

        return true;
    }
    bool isOnly_0(string s) {
        int n = s.length();
        for(int i = 0; i < n; i++) {
            if(s[i] != '0') {
                return false;
            }
        }

        return true;
    }
    int minChanges(string s) {
        int n = s.length();
        int result = 0;

        for(int i = 0; i < n; i += 2) {
            if(s[i] != s[i + 1]) {
                result++;
            }
        }

        return result;
    }
};````
