//appraoch - 3
//time complexity - O(n)
//space complexity - O(1)
class Solution {
public:
    int minimumLength(string s) {
        //base case
        if(s.length() <= 2) {
            return s.length();
        }

        int result = 0;
        vector<int> freq(26, 0);
        for(char &ch: s) {
            freq[ch - 'a']++;
        }

        for(int i = 0; i < 26; i++) {
            if(freq[i] == 0) { //if freq 0, just ignore it
                continue;
            }
            if(freq[i] % 2 == 0) { //if freq is even, add 2 to result
                result += 2;
            }
            else {
                result += 1; //if freq is odd, add 1 to result
            }
        }

        return result;      
    }
};
